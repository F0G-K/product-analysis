### 3.2 组件结构模板

```typescript
// 1. 导入
import { type FC, useState, useCallback } from 'react';
import { cn } from '@/utils/cn';

// 2. 类型定义
interface ButtonProps {
  /** 按钮变体 */
  variant?: 'primary' | 'secondary' | 'danger' | 'ghost';
  /** 尺寸 */
  size?: 'sm' | 'md' | 'lg';
  /** 加载状态 */
  loading?: boolean;
  /** 禁用状态 */
  disabled?: boolean;
  children: React.ReactNode;
  onClick?: () => void;
}

// 3. 组件实现
export const Button: FC<ButtonProps> = ({
  variant = 'primary',
  size = 'md',
  loading = false,
  disabled = false,
  children,
  onClick,
}) => {
  // 4. Hooks
  // 5. 事件处理
  const handleClick = useCallback(() => {
    if (loading || disabled) return;
    onClick?.();
  }, [loading, disabled, onClick]);

  // 6. 渲染
  return (
    <button
      className={cn(
        'inline-flex items-center justify-center rounded-lg font-medium transition-colors',
        'focus:outline-none focus:ring-2 focus:ring-offset-2',
        // variant styles
        variant === 'primary' && 'bg-blue-600 text-white hover:bg-blue-700',
        variant === 'secondary' && 'bg-gray-100 text-gray-900 hover:bg-gray-200',
        // size styles
        size === 'sm' && 'h-8 px-3 text-sm',
        size === 'md' && 'h-10 px-4 text-sm',
        size === 'lg' && 'h-12 px-6 text-base',
        // state styles
        (disabled || loading) && 'opacity-50 cursor-not-allowed',
      )}
      disabled={disabled || loading}
      onClick={handleClick}
    >
      {loading && <Spinner className="mr-2 h-4 w-4" />}
      {children}
    </button>
  );
};
```

### 3.3 组件设计原则

| 原则 | 说明 |
| --- | --- |
| 单一职责 | 一个组件只做一件事，避免 300 行以上的单文件组件 |
| Props 优先 | 通过 Props 控制行为，避免内部硬编码 |
| 组合优于继承 | 使用 `children` 或 slots 组合，不使用类继承 |
| 受控/非受控 | 表单组件支持 `value`/`onChange`（受控）和 `defaultValue`（非受控）两种模式 |
| 类型导出 | 组件的 Props 类型必须 `export`，方便复用 |
| 注释 | Props 使用 JSDoc `/** 说明 */` 注释，生成 IDE 提示 |

### 3.4 状态管理策略

| 状态类型 | 管理方式 | 示例 |
| --- | --- | --- |
| 服务端数据 | React Query / SWR 缓存 | 任务列表、需求详情 |
| 全局应用状态 | Zustand store | 认证信息、当前项目、通知计数 |
| UI 状态 | 组件内 `useState` | 弹窗开关、Tab 切换、表单输入 |
| URL 状态 | React Router search params | 分页、筛选、排序 |

## 4. 状态管理

### 4.1 Zustand Store 模板

```typescript
// stores/authStore.ts
import { create } from 'zustand';
import { persist } from 'zustand/middleware';
import type { User } from '@/types/user';

interface AuthState {
  user: User | null;
  accessToken: string | null;
  isAuthenticated: boolean;

  login: (user: User, token: string) => void;
  logout: () => void;
  setUser: (user: User) => void;
}

export const useAuthStore = create<AuthState>()(
  persist(
    (set) => ({
      user: null,
      accessToken: null,
      isAuthenticated: false,

      login: (user, token) =>
        set({ user, accessToken: token, isAuthenticated: true }),

      logout: () =>
        set({ user: null, accessToken: null, isAuthenticated: false }),

      setUser: (user) => set({ user }),
    }),
    {
      name: 'auth-storage',
      partialize: (state) => ({
        accessToken: state.accessToken,
        isAuthenticated: state.isAuthenticated,
      }),
    }
  )
);
```

### 4.2 服务端数据获取

对于服务端数据（任务列表、需求列表等），优先使用 React Query 进行缓存和自动刷新：

```typescript
// 模式示例
import { useQuery } from '@tanstack/react-query';
import { getTaskDetail } from '@/services/assessment';

export function useTaskDetail(taskId: string) {
  return useQuery({
    queryKey: ['task', 'assessment', taskId],
    queryFn: () => getTaskDetail(taskId),
    enabled: !!taskId,
    staleTime: 30_000,      // 30 秒内视为新鲜
    refetchInterval: 15_000, // 分析中状态每 15 秒轮询
  });
}
```

```typescript
// 认证守卫
function AuthGuard({ children }: { children: React.ReactNode }) {
  const isAuthenticated = useAuthStore((s) => s.isAuthenticated);
  const location = useLocation();

  if (!isAuthenticated) {
    return <Navigate to="/login" state={{ from: location }} replace />;
  }
  return <>{children}</>;
}

// 项目上下文守卫（确保选中项目）
function ProjectGuard({ children }: { children: React.ReactNode }) {
  const { projectId } = useParams();
  const setCurrentProject = useProjectStore((s) => s.setCurrentProject);

  useEffect(() => {
    if (projectId) setCurrentProject(projectId);
  }, [projectId, setCurrentProject]);

  return <>{children}</>;
}
```

## 6. API 集成规范

### 6.1 Axios 实例配置

```typescript
// services/api.ts
import axios, { type AxiosError, type InternalAxiosRequestConfig } from 'axios';
import { useAuthStore } from '@/stores/authStore';

const api = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,  // /api/v1
  timeout: 30_000,
  headers: { 'Content-Type': 'application/json' },
});

// 请求拦截器：自动附加 Token
api.interceptors.request.use((config: InternalAxiosRequestConfig) => {
  const token = useAuthStore.getState().accessToken;
  if (token && config.headers) {
    config.headers.Authorization = `Bearer ${token}`;
  }
  return config;
});

// 响应拦截器：统一错误处理
api.interceptors.response.use(
  (response) => response,
  (error: AxiosError<ApiErrorResponse>) => {
    if (error.response?.status === 401) {
      // Token 过期，清除状态并跳转登录
      useAuthStore.getState().logout();
      window.location.href = '/login';
    }
    return Promise.reject(error);
  }
);

export default api;
```

### 6.4 API 调用原则

| 原则 | 说明 |
| --- | --- |
| 统一实例 | 所有请求通过 `api.ts` 中的 Axios 实例发起，不直接使用 `fetch` |
| 类型安全 | 每个 API 函数声明返回值类型，利用 TypeScript 做编译期校验 |
| 错误传播 | Service 层不做 catch，错误统一由调用方或全局拦截器处理 |
| 取消请求 | 页面卸载时通过 `AbortController` 取消未完成的请求 |
| 加载状态 | 使用 React Query 的 `isLoading` / `isFetching`，避免手动管理 loading |

```typescript
// 分片上传示意
async function uploadLargeFile(file: File, onProgress: (pct: number) => void) {
  const CHUNK_SIZE = 5 * 1024 * 1024; // 5MB
  const chunks = Math.ceil(file.size / CHUNK_SIZE);
  for (let i = 0; i < chunks; i++) {
    const chunk = file.slice(i * CHUNK_SIZE, (i + 1) * CHUNK_SIZE);
    await uploadChunk(chunk, i, chunks);
    onProgress(((i + 1) / chunks) * 100);
  }
  return completeUpload(file.name, chunks);
}
```

## 10. 安全规范

### 10.1 认证与令牌

| 规范 | 说明 |
| --- | --- |
| Token 存储 | `access_token` 仅存内存（Zustand persist 加密存储于 localStorage），不用 `document.cookie` 直接操作 |
| 请求附带 | 通过 Axios 拦截器自动注入 `Authorization: Bearer` 头 |
| Token 过期 | 401 响应自动清除 Token 并跳转登录 |
| Session Cookie | 依赖服务端 `HttpOnly; Secure; SameSite=Lax` Cookie，前端不操作 |

### 10.2 XSS 防护

- React 默认 JSX 转义（`{userInput}` 自动转义），使用 `dangerouslySetInnerHTML` 需经 DOMPurify 清洗
- Markdown 渲染（证据、归因说明）使用 `marked` + `DOMPurify.sanitize()`
- 禁止直接拼接 HTML 字符串到 DOM

### 10.3 CSRF 防护

- 所有状态变更请求依赖 `Authorization: Bearer` 头认证
- Session Cookie 设置为 `SameSite=Lax`，浏览器自动阻止跨站请求携带

### 10.4 敏感数据

- 前端不缓存敏感数据（密码、MFA 密钥、恢复码）
- 恢复码展示后提示用户安全保存，页面关闭后不再可查看
- 文件下载 URL 使用服务端预签名，前端不暴露存储路径
